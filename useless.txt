def rendertext(self):
        """Affiche le plateau complet avec numérotation et les pions pour chaque point"""
        board_str = "\n--- Plateau de Backgammon ---\n"
        for i in range(24):
            point = i + 1
            p1 = self.board[i, 0]
            p2 = self.board[i, 1]
            board_str += f"Point {point:2d} | Joueur 1: {p1:2d} | Joueur 2: {p2:2d}\n"
        print(board_str)

    def help_map(self):
        """Affiche une aide avec la numérotation du plateau"""
        map_str = "\n=== Carte du plateau ===\n"
        map_str += "Les points sont numérotés de 1 à 24.\n"
        map_str += "Pour Joueur 1, les pions se déplacent vers les points de plus petite valeur (et sortent en 0).\n"
        map_str += "Pour Joueur 2, les pions se déplacent vers les points de plus grande valeur (et sortent en 25).\n"
        map_str += "========================\n"
        print(map_str)

if __name__ == '__main__':
    env = BackgammonEnv()
    env.reset()
    print("=== Bienvenue au Backgammon ===")
    print("Tape 'map' à tout moment pour afficher la carte du plateau.")
    while True:
        dice = env.roll_dice()
        remaining_dice = dice.copy()
        print(f"\nJoueur {env.current_player + 1}, c'est ton tour !")
        print(f"Résultat des dés : {remaining_dice}")

        # Boucle tant qu'il reste des dés utilisables
        while remaining_dice:
            valid = env.valid_moves(remaining_dice)
            if not valid:
                print("Aucun mouvement possible avec les dés restants.")
                break
            print("\nVoici les mouvements possibles (format: [source, destination, dé utilisé]):")
            for m in valid:
                print(m)
            print("\nTape 'map' pour afficher le plateau.")
            print("Tape 'pass' pour passer ton tour.")
            user_input = input("Mouvement (ex: 12,7,4 pour déplacer de 12 à 7 en utilisant 4) : ").strip()
            if user_input.lower() == "pass":
                break
            if user_input.lower() == "map":
                from backgammon_gui import show_map_tkinter
                show_map_tkinter(env)
                #env.render()
                continue
            try:
                parts = user_input.split(",")
                if len(parts) != 3:
                    print("Format invalide. Utilise source,destination,dé utilisé.")
                    continue
                src = int(parts[0])
                dest = int(parts[1])
                die_val = int(parts[2])
            except:
                print("Erreur dans la saisie. Réessaie.")
                continue

            # Vérification que la combinaison de dés correspond au déplacement demandé
            subset = find_subset(remaining_dice, die_val)
            if subset is None:
                print(f"Aucune combinaison de dés dans {remaining_dice} ne donne {die_val}.")
                continue

            # Vérification que le mouvement est dans la liste des mouvements valides
            if (src, dest, die_val) not in valid:
                print("Ce mouvement n'est pas valide. Réessaie.")
                continue

            success, win = env.step_move(src, dest, die_val)
            if not success:
                print("Mouvement impossible. Réessaie.")
                continue

            # Retirer les dés utilisés (peut être une combinaison de plusieurs dés)
            for die in subset:
                remaining_dice.remove(die)
            env.render()
            print("Historique des coups :\n", env.historique.to_string(index=False))
            if win:
                print(f"Félicitations, Joueur {env.current_player + 1} a gagné !")
                exit()
        # Fin du tour, changement de joueur
        env.current_player = 1 - env.current_player
